    70	void cPlaceItDrawer::drawStatusOfStructureAtCell(cBuildingListItem *itemToPlace, int mouseCell)
    71	{
    72	    d2tm_assert(itemToPlace);
    73	    if (mouseCell < 0) return;
    74	    Color itemToPlaceColor;
    75	
    76	    int structureId = itemToPlace->getBuildId();
    77	    d2tm_assert(structureId > -1);
    78	
    79	    bool bWithinBuildDistance = false;
    80	    int cellWidth = m_structureUtils->getWidthOfStructureTypeInCells(structureId);
    81	    int cellHeight = m_structureUtils->getHeightOfStructureTypeInCells(structureId);
    82	
    83	#define SCANWIDTH	1
    84	
    85	    int iCellX = m_objects->getMapGeometry()->getCellX(mouseCell);
    86	    int iCellY = m_objects->getMapGeometry()->getCellY(mouseCell);
    87	
    88	    // check
    89	    int iStartX = iCellX-SCANWIDTH;
    90	    int iStartY = iCellY-SCANWIDTH;
    91	
    92	    int iEndX = iCellX + SCANWIDTH + cellWidth;
    93	    int iEndY = iCellY + SCANWIDTH + cellHeight;
    94	
    95	    // Fix up the boundaries
    96	    cPoint::split(iStartX, iStartY) = m_objects->getMapGeometry()->fixCoordinatesToBeWithinMap(iStartX, iStartY);
    97	    cPoint::split(iEndX, iEndY) = m_objects->getMapGeometry()->fixCoordinatesToBeWithinMap(iEndX, iEndY);
    98	
    99	    // Determine if structure to be placed is within build distance
   100	    for (int iX=iStartX; iX < iEndX; iX++) {
   101	        for (int iY=iStartY; iY < iEndY; iY++) {
   102	            int iCll = m_objects->getMapGeometry()->getCellWithMapDimensions(iX, iY);
   103	
   104	            if (iCll > -1) {
   105	                int idOfStructureAtCell = m_objects->getMap()->getCellIdStructuresLayer(iCll);
   106	                if (idOfStructureAtCell > -1) {
   107	                    int iID = idOfStructureAtCell;
   108	
   109	                    if (m_objects->getStructures()[iID]->getOwner() == m_player->getId()) {
   110	                        bWithinBuildDistance = true; // connection!
   111	                        break;
   112	                    }
   113	
   114	                    // TODO: Allow placement nearby allies?
   115	
   116	                    // structure owned by someone else: its foundation slab must not
   117	                    // count as a build anchor, so skip the wall/slab check for this cell
   118	                    continue;
   119	                }
   120	
   121	                if (m_objects->getMap()->getCellType(iCll) == TERRAIN_WALL ||
   122	                        m_objects->getMap()->getCellType(iCll) == TERRAIN_SLAB) {
   123	                    bWithinBuildDistance=true;
   124	                    // TODO: here we should actually find out if the slab is ours or not??
   125	                    break;
   126	                }
   127	            }
   128	        }
   129	    }
   130	
   131	    int iDrawX = m_objects->getMap()->mouse_draw_x(mouseCell);
   132	    int iDrawY = m_objects->getMap()->mouse_draw_y(mouseCell);
   133	
   134	    if (!bWithinBuildDistance) {
   135	        itemToPlaceColor = Color::PlaceNeutral;
   136	    }
   137	    else {
   138	        // Draw over it the mask for good/bad placing (decorates temp bitmap)
   139	        for (int iX=0; iX < cellWidth; iX++) {
   140	            for (int iY=0; iY < cellHeight; iY++) {
   141	                itemToPlaceColor = Color::PlaceNeutral;
   142	
   143	                int cellX = iCellX + iX;
   144	                int cellY = iCellY + iY;
   145	
   146	                if (!m_objects->getMapGeometry()->isWithinBoundaries(cellX, cellY)) {
   147	                    continue;
   148	                }
   149	
   150	                int iCll = m_objects->getMapGeometry()->makeCell(cellX, cellY);
   151	
   152	                if (!m_objects->getMap()->isCellPassable(iCll) || m_objects->getMap()->getCellType(iCll) != TERRAIN_ROCK) {
   153	                    itemToPlaceColor = Color::PlaceBad;
   154	                }
   155	
   156	                if (m_objects->getMap()->getCellType(iCll) == TERRAIN_SLAB) {
   157	                    itemToPlaceColor = Color::PlaceGood;
   158	                }
   159	
   160	                // occupied by units or structures
   161	                int idOfStructureAtCell = m_objects->getMap()->getCellIdStructuresLayer(iCll);
   162	                if (idOfStructureAtCell > -1) {
   163	                    itemToPlaceColor = Color::PlaceBad;
   164	                }
   165	
   166	                int unitIdOnMap = m_objects->getMap()->getCellIdUnitLayer(iCll);
   167	                if (unitIdOnMap > -1) {
   168	                    // temporarily dead units do not block, but alive units (non-dead) do block placement
   169	                    if (!m_objects->getUnit(unitIdOnMap)->isDead()) {
   170	                        itemToPlaceColor = Color::PlaceBad;
   171	                    }
   172	                    // TODO: Allow placement, let units move aside when clicking before placement?
   173	                }
   174	
   175	                // Draw bad gfx on spot
   176	                float desiredWidth = m_mapCamera->getZoomedTileWidth();
   177	                float desiredHeight = m_mapCamera->getZoomedTileHeight();
   178	                float posX = iX * desiredWidth;
   179	                float posY = iY * desiredHeight;
   180	                // cRectangle rectangle = cRectangle(posX, posY, desiredWidth, desiredHeight);
   181	                m_sdlDrawer->renderRectFillColor(iDrawX+posX, iDrawY+posY, desiredWidth, desiredHeight,itemToPlaceColor);
   182	            }
   183	        }
   184	    }

   187	void cPlaceItDrawer::drawStructureIdAtMousePos(cBuildingListItem *itemToPlace, int mouseCell)
   188	{
   189	    d2tm_assert(itemToPlace);
   190	
   191	    int structureId = itemToPlace->getBuildId();
   192	
   193	    int iDrawX = m_objects->getMap()->mouse_draw_x(mouseCell);
   194	    int iDrawY = m_objects->getMap()->mouse_draw_y(mouseCell);
   195	
   196	    int width = m_infos->getStructureInfo(structureId).bmp_width;
   197	    int height = m_infos->getStructureInfo(structureId).bmp_height;
   198	
   199	    int scaledWidth = m_mapCamera->factorZoomLevel(width);
   200	    int scaledHeight = m_mapCamera->factorZoomLevel(height);
   201	
   202	    Texture *bmp = nullptr;
   203	    if (structureId == SLAB1) {
   204	        bmp = m_gfxdata->getTexture(PLACE_SLAB1);
   205	    }
   206	    else if (structureId == SLAB4) {
   207	        bmp = m_gfxdata->getTexture(PLACE_SLAB4);
   208	    }
   209	    else if (structureId == WALL) {
   210	        bmp = m_gfxdata->getTexture(PLACE_WALL);
   211	    }
   212	    else {
   213	        bmp = m_player->getStructureBitmap(structureId);
   214	    }
   215	    // We need to take a source rectangle, as the structure BMP contains multiple frames of the structure
   216	    // therefore we cannot use renderStrechFullSprite (as that assumes the source texture can be used as a whole
   217	    // which cannot be used in this case)
   218	    cRectangle src = { 0, 0, width, height}; // takes first frame
   219	    cRectangle dest= {iDrawX, iDrawY, scaledWidth, scaledHeight};
   220	    m_sdlDrawer->renderStrechSprite(bmp, src, dest,96);
   221	}
