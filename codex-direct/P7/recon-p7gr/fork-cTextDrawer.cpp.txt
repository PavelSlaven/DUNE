    32	bool cTextDrawer::shouldUseFallback(const std::string &msg) const
    33	{
    34	    if (!m_fallbackFont) return false;
    35	    const auto found = m_fallbackByText.find(msg);
    36	    if (found != m_fallbackByText.end()) return found->second;
    37	
    38	    const bool useFallback = !ttfFontSupportsText(m_font, msg) &&
    39	        ttfFontSupportsText(m_fallbackFont, msg);
    40	    m_fallbackByText.emplace(msg, useFallback);
    41	    return useFallback;
    42	}
    43	
    44	TTF_Font *cTextDrawer::fontFor(const std::string &msg) const
    45	{
    46	    return shouldUseFallback(msg) ? m_fallbackFont : m_font;
    47	}

    49	cTextTextureCache *cTextDrawer::cacheFor(const std::string &msg) const
    50	{
    51	    return shouldUseFallback(msg) ? m_fallbackTextCache.get() : m_textCache.get();
    52	}

    54	void cTextDrawer::drawText(int x, int y, Color color, const std::string &msg, bool applyShadow) const
    55	{
    56	    if (msg.empty()) return;
    57	    auto cacheEntry = cacheFor(msg)->findOrCreate(color, msg);
    58	    cacheEntry->lifeCounter += 1;
    59	    if (uiaudit::isEnabled()) {
    60	        uiaudit::recordText(msg, x, y, cacheEntry->width, cacheEntry->height);
    61	    }
    62	    if (applyShadow) {
    63	        renderDrawer->renderTexture(cacheEntry->shadowsTexture, x + 1, y + 1,cacheEntry->width, cacheEntry->height);
    64	    }
    65	    renderDrawer->renderTexture(cacheEntry->texture, x, y,cacheEntry->width, cacheEntry->height);
    66	}

    83	void cTextDrawer::drawTextCentered(const std::string &msg, int y, Color color) const
    84	{
    85	    if (msg.empty()) return;
    86	    int w,h;
    87	    TTF_SizeUTF8(fontFor(msg), msg.c_str(), &w, &h);
    88	    w = (w + m_density - 1) / m_density;
    89	    int half = w / 2;
    90	    int xPos = (game.m_screenW / 2) - half;
    91	    drawText(xPos, y, color, msg);
    92	}

    94	void cTextDrawer::drawTextCenteredInBox(const std::string &msg, int x, int y, int boxWidth, int boxHeight, Color color) const
    95	{
    96	    if (msg.empty()) return;
    97	    int w,h;
    98	    TTF_SizeUTF8(fontFor(msg), msg.c_str(), &w, &h);
    99	    w = (w + m_density - 1) / m_density;
   100	    h = (h + m_density - 1) / m_density;
   101	    int lenghtInPixels = w;
   102	    int heightInPixels = h;
   103	
   104	    int halfLengthInPixels = lenghtInPixels / 2;
   105	    int xPos = x + ((boxWidth / 2) - halfLengthInPixels);
   106	    int halfHeightInPixels = heightInPixels / 2;
   107	    int yPos = y + ((boxHeight / 2) - halfHeightInPixels);
   108	    std::optional<uiaudit::BoxScope> auditBox;
   109	    if (uiaudit::isEnabled()) {
   110	        auditBox.emplace(x, y, boxWidth, boxHeight);
   111	    }
   112	    drawText(xPos, yPos, color, msg);
   113	}

   125	void cTextDrawer::drawTextScaledCenteredInBox(
   126	    const std::string &msg,
   127	    const cRectangle &rect,
   128	    float scale,
   129	    Color color,
   130	    bool applyShadow) const
   131	{
   132	    if (msg.empty()) return;
   133	    scale = std::max(0.5F, scale);
   134	    auto cacheEntry = cacheFor(msg)->findOrCreate(color, msg);
   135	    cacheEntry->lifeCounter += 1;
   136	    const int width = static_cast<int>(cacheEntry->width * scale + 0.5F);
   137	    const int height = static_cast<int>(cacheEntry->height * scale + 0.5F);
   138	    const int x = rect.getX() + (rect.getWidth() - width) / 2;
   139	    const int y = rect.getY() + (rect.getHeight() - height) / 2;
   140	    if (uiaudit::isEnabled()) {
   141	        const uiaudit::Rect box{rect.getX(), rect.getY(), rect.getWidth(), rect.getHeight()};
   142	        uiaudit::recordText(msg, x, y, width, height, &box);
   143	    }
   144	    if (applyShadow) {
   145	        renderDrawer->renderTexture(cacheEntry->shadowsTexture, x + 2, y + 2, width, height);
   146	    }
   147	    renderDrawer->renderTexture(cacheEntry->texture, x, y, width, height);
   148	}

   150	void cTextDrawer::drawTextScaledOpticallyCenteredInBox(
   151	    const std::string &msg,
   152	    const cRectangle &rect,
   153	    float scale,
   154	    Color color,
   155	    bool applyShadow) const
   156	{
   157	    if (msg.empty()) return;
   158	    scale = std::max(0.5F, scale);
   159	    auto cacheEntry = cacheFor(msg)->findOrCreate(color, msg);
   160	    cacheEntry->lifeCounter += 1;
   161	    const int width = static_cast<int>(cacheEntry->width * scale + 0.5F);
   162	    const int height = static_cast<int>(cacheEntry->height * scale + 0.5F);
   163	    const cPoint origin = calculateOpticallyCenteredTextOrigin(
   164	        rect,
   165	        width,
   166	        height,
   167	        cacheEntry->inkBounds
   168	    );
   169	    if (uiaudit::isEnabled()) {
   170	        const uiaudit::Rect box{rect.getX(), rect.getY(), rect.getWidth(), rect.getHeight()};
   171	        uiaudit::recordText(msg, origin.x, origin.y, width, height, &box);
   172	    }
   173	    if (applyShadow) {
   174	        renderDrawer->renderTexture(
   175	            cacheEntry->shadowsTexture,
   176	            origin.x + 2,
   177	            origin.y + 2,
   178	            width,
   179	            height
   180	        );
   181	    }
   182	    renderDrawer->renderTexture(cacheEntry->texture, origin.x, origin.y, width, height);
   183	}

   186	{

   223	int cTextDrawer::getFontHeight() const
   224	{
   225	    const int primaryHeight = TTF_FontHeight(m_font);
   226	    const int fallbackHeight = m_fallbackFont ? TTF_FontHeight(m_fallbackFont) : 0;
   227	    return (std::max(primaryHeight, fallbackHeight) + m_density - 1) / m_density;
   228	}

   241	{
